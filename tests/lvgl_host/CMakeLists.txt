cmake_minimum_required(VERSION 3.18)
project(paper_render_test C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDK}/managed_components/lvgl__lvgl" lvgl)
set_property(SOURCE "${SDK}/managed_components/lvgl__lvgl/src/draw/sw/blend/lv_draw_sw_blend_to_i1.c"
    TARGET_DIRECTORY lvgl APPEND PROPERTY COMPILE_OPTIONS "-include${SDK}/main/ui/lvgl_i1_override.h")
file(GLOB LAB_FONTS "${SDK}/main/font_lab_assets/*.c")
add_executable(paper_render_test main.cpp "${SDK}/main/ui_demo.cc" "${SDK}/main/ui_demo_fonts.c" ${LAB_FONTS})
target_include_directories(paper_render_test PRIVATE "${SDK}/main")
target_compile_options(paper_render_test PRIVATE -UNDEBUG)
target_link_libraries(paper_render_test PRIVATE lvgl)
