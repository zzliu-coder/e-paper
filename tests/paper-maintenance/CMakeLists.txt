cmake_minimum_required(VERSION 3.16)
project(paper_maintenance_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(GLOB CORE "${SDK}/components/paper_core/src/*.cpp")
add_library(core STATIC ${CORE} "${SDK}/components/paper_core/vendor/tjpgd/tjpgd.c")
set(CROSSMUX "${SDK}/components/paper_reader_crossmux")
include("${CROSSMUX}/sources.cmake")
add_library(crossmux STATIC ${CROSSMUX_SOURCES})
target_include_directories(crossmux PUBLIC ${CROSSMUX_INCLUDES})
target_compile_definitions(crossmux PRIVATE ${CROSSMUX_DEFINITIONS})
target_compile_features(crossmux PUBLIC cxx_std_20)
target_link_libraries(core PUBLIC crossmux)
target_include_directories(core PUBLIC "${SDK}/components/paper_core/include")
target_include_directories(core PRIVATE "${SDK}/managed_components/lvgl__lvgl/src/libs/gltf/stb_image")
find_package(ZLIB REQUIRED)
target_link_libraries(core PUBLIC ZLIB::ZLIB)
target_compile_options(core PRIVATE -Wall -Wextra -UNDEBUG)
add_executable(maintenance_test main.cpp)
target_link_libraries(maintenance_test PRIVATE core)
target_compile_options(maintenance_test PRIVATE -UNDEBUG)
add_executable(performance_test performance.cpp)
target_link_libraries(performance_test PRIVATE core)
target_compile_options(performance_test PRIVATE -UNDEBUG)
add_executable(reader_content_test reader_content.cpp)
add_executable(crossmux_test crossmux.cpp)
add_executable(crossmux_images_test crossmux_images.cpp)
add_executable(image_decode_test image_decode.cpp)
add_executable(font_proof_test font_proof.cpp)
target_link_libraries(font_proof_test PRIVATE core)
target_compile_options(font_proof_test PRIVATE -UNDEBUG)
target_link_libraries(image_decode_test PRIVATE core)
target_compile_options(image_decode_test PRIVATE -UNDEBUG)
target_link_libraries(crossmux_images_test PRIVATE core ${CMAKE_DL_LIBS})
target_compile_options(crossmux_images_test PRIVATE -UNDEBUG)
target_link_libraries(crossmux_test PRIVATE core ${CMAKE_DL_LIBS})
target_compile_options(crossmux_test PRIVATE -UNDEBUG)
target_link_libraries(reader_content_test PRIVATE core)
target_compile_options(reader_content_test PRIVATE -UNDEBUG)
add_executable(font_candidate_test font_candidate.cpp)
target_link_libraries(font_candidate_test PRIVATE core)
target_compile_options(font_candidate_test PRIVATE -UNDEBUG)
add_executable(offline_test offline.cpp)
target_link_libraries(offline_test PRIVATE core)
target_compile_options(offline_test PRIVATE -UNDEBUG)
add_executable(bluetooth_test bluetooth.cpp "${SDK}/main/paper_shell/bluetooth_service.cc")
target_include_directories(bluetooth_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/bt_stubs" "${SDK}/main/paper_shell")
target_link_libraries(bluetooth_test PRIVATE core)
target_compile_options(bluetooth_test PRIVATE -UNDEBUG)
add_executable(refresh_test refresh.cpp "${SDK}/main/paper_shell/gray/driver.c")
